\section{Introduction}

\begin{problem}
    Determine which of the following binary operations are associative:
    \begin{enumerate}
        \item the operation on $\ZZ$ defined by $a*b = a - b$
        \item the operation on $\RR$ defined by $a*b = a + b + ab$
        \item the operation on $\QQ$ defined by $a*b = \frac{a+b}{5}$
        \item the operation on $\ZZ \times \ZZ$ defined by $(a,b)*(c,d) = (ad+bc, bd)$
        \item the operation on $\QQ-\{0\}$ defined by $a*b = \frac{a}{b}$
    \end{enumerate}
\end{problem}

\begin{proof}
    This exercise is pretty trivial as we just have to compute $a*(b*c)=(a*b)*c$.
    For $1.$, $a*(b*c)=a*(b-c)=a-(b-c)=a-b+c$ but $(a*b)*c=(a-b)*c=(a-b)-c=a-b-c$.
    Rest of them are pretty similar.
\end{proof}

\begin{problem}
    Prove that addition of residue classes in $\ZZ/n\ZZ$ is associative.
\end{problem}

\begin{proof}
    By definition of residue addition,
    \[ [a]+[b]= [a+b] \]
    Therefore, we get the following.
    \begin{align*}
        [a]+([b]+[c])&=[a]+[b+c]\\
        &= [a+b+c] \\
        &= [a+b]+[c] \\
        &= ([a]+[b])+[c]
    \end{align*}
\end{proof}
