\begin{problem}
    Determine which of the following binary operations are associative:
    \begin{enumerate}
        \item the operation on $\ZZ$ defined by $a*b = a - b$
        \item the operation on $\RR$ defined by $a*b = a + b + ab$
        \item the operation on $\QQ$ defined by $a*b = \frac{a+b}{5}$
        \item the operation on $\ZZ \times \ZZ$ defined by $(a,b)*(c,d) = (ad+bc, bd)$
        \item the operation on $\QQ-\{0\}$ defined by $a*b = \frac{a}{b}$
    \end{enumerate}
\end{problem}

\begin{proof}
    This exercise is pretty trivial as we just have to compute $a*(b*c)=(a*b)*c$.
    For $1.$, $a*(b*c)=a*(b-c)=a-(b-c)=a-b+c$ but $(a*b)*c=(a-b)*c=(a-b)-c=a-b-c$.
    Rest of them are pretty similar.
\end{proof}

\begin{problem}
    Prove that addition of residue classes in $\ZZ/n\ZZ$ is associative.
\end{problem}

\begin{proof}
    By definition of residue addition,
    \[ [a]+[b]= [a+b] \]
    Therefore, we get the following.
    \begin{align*}
        [a]+([b]+[c])&=[a]+[b+c]\\
        &= [a+(b+c)] \\
        &= [(a+b)+c] \\
        &= [a+b]+[c] \\
        &= ([a]+[b])+[c]
    \end{align*}
\end{proof}

\begin{problem}
    Prove that multiplication of residue classes in $\ZZ/n\ZZ$ is associative.
\end{problem}

\begin{proof}
     By definition of residue multiplication,
    \[ [a]\cdot[b]= [a \cdot b] \]
    Therefore, we get the following.
    \begin{align*}
        [a]\cdot ([b] \cdot [c]) &= [a]\cdot [b \cdot c] \\
        &= [a \cdot (b \cdot c) ] \\
        &= [(a \cdot b) \cdot c] \\
        &= [a \cdot b] \cdot [c] \\
        &= ([a] \cdot [b]) \cdot [c]
    \end{align*}
\end{proof}

\begin{problem}
    Prove for all $n > 1$ that $\ZZ/n\ZZ$ is not a group under multiplication of residue classes.
\end{problem}

\begin{proof}
    There is no multiplicative inverse for $[a]$ with $\gcd(a,n)>1$. The proof for it pretty trivial,
    if there was a inverse, it would mean that
    \[ a \cdot b \equiv 1 \pmod{n} \implies 0 \equiv 1 \pmod{\gcd(a,n)} \]
    That's a contradiction. 
\end{proof}

\begin{problem}
    Determine which of the following sets are groups under addition:
    \begin{enumerate}
        \item the set of rational numbers (including $0 = 0/1$) in lowest terms whose denominators are odd
        \item the set of rational numbers (including $0 = 0/1$) in lowest terms whose denominators are even
        \item the set of rational numbers of absolute value $< 1$
        \item the set of rational numbers of absolute value $\ge 1$ together with 0
        \item the set of rational numbers with denominators equal to 1 or 2
        \item the set of rational numbers with denominators equal to 1, 2 or 3.
    \end{enumerate}
\end{problem}

\begin{proof}
    Below are the answers: 
    \begin{enumerate}
        \item Group.
        \item Not a group. Because it is not closed. (Eg: $5/2 + 7/2 = 6$)
        \item Not a group. Because again it is not closed. (Eg : $0.99+0.98=1.97$)
        \item Not a group. Because again it is not closed. (Eg : $4+(-3.1)=0.9$)
        \item Group.
        \item Not a group. Because its not closed. (Eg : $1/2+1/3=5/6$)
    \end{enumerate}
\end{proof}

\begin{problem}
    Let $G = \{x \in \RR \mid 0 \le x < 1\}$ and for $x, y \in G$ let $x*y$ be the fractional part of $x+y$ 
    (i.e., $x*y = x+y - \floor{x+y}$ where $\floor{a}$ is the greatest integer less than or equal to $a$).
    Prove that $*$ is a well defined binary operation on $G$ and that $G$ is an abelian group under $*$ (called the real numbers mod 1).
\end{problem}

\begin{proof}
    It is well defined as,
    \[ x_1 = x_2, y_1 = y_2 \implies x_1*y_1 = x_1+y_1-\floor{x_1+y_1}\]
    \[ \implies x_1+y_1-\floor{x_1+y_1}=x_2+y_2 - \floor{x_2+y_2} = x_2 * y_2 \]
    Also, its closed as $x*y = \{x+y\}$ and $0 \le \{x+y\} < 1$. One can check that inverses and identity exists. For associativity,
    \begin{align*}
        x*(y*z) &= x*(y+z-\floor{y+z}) \\
        &= \{x+y+z - \floor{y+z}\} \\
        &= \{ x+y+z \} \\
        &= \{ x+y-\floor{x+y}+z \} \\
        &= (x*y)*z
    \end{align*}
\end{proof}

\begin{problem}
    Let $G = \{z \in \CC \mid z^n = 1 \text{ for some } n \in \ZZ^+\}$.
    \begin{enumerate}
        \item Prove that $G$ is a group under multiplication (called the group of roots of unity in $\CC$).
        \item Prove that $G$ is not a group under addition.
    \end{enumerate}
\end{problem}

\begin{proof}
    For $G$ under multiplication, we know that for all $z_1,z_2,z_3 \in G$ we have
    \[ z_1 \cdot (z_2 \cdot z_3) = (z_1 \cdot z_2) \cdot z_3 \]
    Also, it is closed as for all $z_1, z_2 \in G$ we have $z_1^n=1$ and $z_2^k=1$ so 
    \[ (z_1 \cdot z_2)^{\text{lcm}(n,k)}=1 \]
    Now, for inverse of a $z \in G$ just take $1/z$. As $(1/z)^n=1/1=1$ so $1/z \in G$. Also, $z \cdot 1/z = 1$.

    For $G$ under addition, if it was a group then, $1 \in G$ as $1^1 = 1 \in G \implies 1+1 = 2 \in G$ but there does not exist any integer
    $n$ such that $2^n=1$.   
\end{proof}

\begin{problem}
    Let $G = \{a + b\sqrt{2} \in \RR \mid a, b \in \QQ\}$.
    \begin{enumerate}
        \item Prove that $G$ is a group under addition.
        \item Prove that the nonzero elements of $G$ are a group under multiplication. 
    \end{enumerate}
\end{problem}

\begin{proof}
    We know reals are associative so the elements of $G$ are associative under addition. The inverse for $a+b \sqrt{2}$ is $(-a)+(-b)\sqrt{2}$.
    It is also closed as,
    \[ a+b\sqrt{2} + c + d\sqrt{2} = (a+c)+(b+d)\sqrt{2} \]
    The identity is going to  be $0+0\cdot \sqrt{2}=0$.

    We know reals are associative so the elements of $G$ are associative under multiplication. 
    The inverse for $a+b \sqrt{2}$ is $\displaystyle \frac{a}{a^2-2b^2} - \frac{b}{a^2-2b^2} \sqrt{2}$.
    The identity is $1+0 \cdot \sqrt{2}=1$. And its closed under multiplication as 
    \[ (a+b\sqrt{2})(c+d\sqrt{2})= (ac+2bd)+(ad+bc)\sqrt{2} \]
\end{proof}

\begin{problem}
    Prove that a finite group is abelian if and only if its group table is a symmetric matrix.
\end{problem}

\begin{proof}
    Let $G = \{g_1,\ldots,g_n\}$ be an group with $n$ elements. Then we define the group matrix to be a $n \times n$ square matrix 
    with entires $a_{ij} = g_i g_j$. So, if its G is abelian then $a_{ij}=g_i g_j = g_j g_i = a_{ji}$, so its symmetric. 
    If the group table is symmetric then $a_{ij}=a_{ji} \implies g_ig_j = g_jg_i$ for all $1 \le i,j \le n$. Thus, its abelian. 
\end{proof}

\begin{problem}
    Prove that $(a_1 a_2 \dots a_n)^{-1} = a_n^{-1} a_{n-1}^{-1} \dots a_1^{-1}$ for all $a_1, a_2, \dots, a_n \in G$.
\end{problem}

\begin{proof}
    We will use strong induction. Its obviously true for $n=1$ and assume its true for $n=2,\ldots,k$. Then,
    \begin{align*}
        ((a_1 \cdots a_{k}) \cdot a_{k+1})^{-1} &= a_{k+1}^{-1} (a_1 \cdots a_k)^{-1} \\
        &= a_{k+1}^{-1} a_k^{-1} \cdots a_1^{-1}
    \end{align*}
\end{proof}

    and if $|x|=2$ then $x^2=1$.
\end{proof}

\begin{problem}
    Let $x$ be an element of $G$. Prove that if $|x| = n$ for some positive integer $n$ then $x^{-1} = x^{n-1}$.
\end{problem}

\begin{proof}
    If $|x|=n$ then $x^n=1 \implies x^{n-1} * x^{1} = 1 \implies x^{n-1} = 1 * x^{-1} = x^{-1}$.
\end{proof}

\begin{problem}
 For $x$ an element in $G$ show that $x$ and $x^{-1}$ have the same order.
\end{problem}

\begin{proof}
    If $|x|= \infty$ then $|x^{-1}|=\infty$ because if $|x^{-1}|$ where to be finite it would lead to a pretty obvious contradiction.
    If $|x|=k$ then $x^k=1 \implies x^{-k}=1$ so $|x^{-1}| \le k$ and if $|x^{-1}| = i < k$ then $x^{-i}=1 \implies x^{i}=1 
    \implies k \le i < k$ which is a contradiction.
\end{proof}

\begin{problem}
    Let $G$ be a finite group and let $x$ be an element of $G$ of order $n$. Prove that if $n$ is odd, then $x = (x^2)^k$ for some $k$.
\end{problem}

\begin{proof}
    Since $n$ is the order of the element $x$, we have $x^n=1 \Rightarrow x^{n+1}=x$. Since $n+1 = 2k$ for some $k \in \ZZ^+$ we have
    \[ x^{2k} = x \Rightarrow (x^2)^k=x \]  
\end{proof}

\begin{problem}
    If $x$ and $g$ are elements of the group $G$, prove that $|x| = |g^{-1}xg|$. Deduce that $|ab| = |ba|$ for all $a, b \in G$.
\end{problem}

\begin{proof}
    We can easily prove through induction that $(g^{-1}xg)^k=g^{-1}x^kg$. Now let $|x|=n$ and let $|g^{-1}xg|=m$ and suppose
    $m < n$ then 
    \[ g^{-1}x^mg = e \Rightarrow x^m=e \Rightarrow |x| = n \le m < n \]
    Thus, $m \ge n$ and $g^{-1}x^ng=g^{-1} e g = e$ so we have $|g^{-1}xg|=n=|x|$.

    Also, if you let $x=ab$ and $g=a$ we get,
    \begin{align*}
        |ab| &= |(a)^{-1} (ab) (a)| \\
         &= |a^{-1}(ab)a| \\
         &= |ba|
    \end{align*}
\end{proof}

\begin{problem}
    Suppose $x \in G$ and $|x| = n < \infty$. If $n = st$ for some positive integers $s$ and $t$, prove that $|x^s| = t$.
\end{problem}

\begin{proof}
    If $|x|=n$ and $n=st$ then $x^n=1 \Rightarrow x^{st}=1 \Rightarrow (x^s)^t=1$. Thus, $|x^s| \le t$.
    Suppose $|x^s|=k<t$ then $x^{sk}=1$ but $n=st \le sk \Rightarrow t \le k < t$. Thus, $|x^s|=t$.
\end{proof}

\begin{problem}
     Prove that $A \times B$ is an abelian group if and only if both $A$ and $B$ are abelian.
\end{problem}

\begin{proof}
    Suppose $A \times B$ is abelian then,
    \begin{align*}
        (a,b)(c,d) &= (c,d)(a,b) \\
        (a * c, b \circ d) &= (c * a,d \circ b) \\
    \end{align*}
    Thus $a*c=c*a$ and $b \circ d = d \circ b$. Therefore, both $A$ and $B$ are abelian.

    And if $A$ and $B$ are abelian then
    \begin{align*}
        (a,b)(c,d) &= (a * c, b \circ d) \\
        &= (c * a, d \circ b) \\
        &= (c,d)(a,b)
    \end{align*}
\end{proof}

\begin{problem}
    Suppose $G$ is a group and $|x|=n < \infty$. If $x^a = 1$ then $n \mid a$. 
\end{problem}

\begin{proof}
    Using the Euclidian algorithm we get,
    \[ a=nq+r \]
    where $0 \le r < n$
    Then,
    \[ x^a = x^{nq+r} = x^{r} = 1 \]
    But that's a contradiction unless $r=0$. 
\end{proof}

\begin{problem}
    Prove that the elements $(a, 1)$ and $(1, b)$ of $A \times B$ commute and deduce that the order of $(a, b)$ is the least common 
    multiple of $|a|$ and $|b|$.
\end{problem}

\begin{proof}
    For the first statement, 
    \[ (a,1)(1,b) = (a*1,1 \circ b) = (1*a,b \circ 1) = (1,b)(a,1) \]
    Now, suppose $|(a,b)|=n$ then,
    \begin{align*}
        (a,b)^n &= (1,1) \\
        \implies (a^n,b^n) &= (1,1)
    \end{align*}
    Thus, $a^n=1$ and $b^n=1$. Therefore, $n \mid |a|$ and $n \mid |b|$. Thus, $n \mid \text{lcm}(|a|,|b|)$ which implies that
    $n \ge \text{lcm}(|a|,|b|)$. Now, $n=\text{lcm}(|a|,|b|)$ satisfies $(a,b)^n=(1,1)$. 
\end{proof}

\begin{problem}
    Prove that any finite group $G$ of even order contains an element of order 2.
\end{problem}

\begin{proof}
    Let $\iota(G)=\{g \in G \mid g \neq g^{-1}\}$. Then notice that order of $\iota(G)$ is even as it contains both $g,g^{-1}$.
    Then, the order of $G-\iota(G)$ is at least $2$ because the order of $G$ is even. Thus, we have a non identity element 
    $x \in G - \iota(G)$ such that $x^2=1$.  
\end{proof}

\begin{problem}
    If $x$ is an element of finite order $n$ in $G$, prove that the elements of
    \[ C=\{x^k \mid 0 \le k \le n-1\} \]
    are all distinct.
    Deduce that $|x| \le |G|$.
\end{problem}

\begin{proof}
    Suppose that for $a \neq b$ we have,
    \[ x^a = x^b \]
    Then, $x^{a-b}=1$ which implies $n \mid a-b$ but $a-b < |a| < n$ which is a contradiction. Thus, all the elements of $C$ are distinct.
    And since $C$ is a subgroup we have 
    \[ |x|=n=|C| \le |G| \]
\end{proof}

        \item Prove that if $n = 2k$ and $1 \le i < n$ then $x^i = x^{-i}$ if and only if $i = k$.

\begin{problem}
    Assume $G = \{1, a, b, c\}$ is a group of order 4 with identity 1. Assume also that $G$ has no elements of order 4.
    Prove that there is a unique group table for $G$. Deduce that $G$ is abelian.
\end{problem}

\begin{proof}
    Notice that $a^2=b^2=c^2=1$, as $|a| \mid |G|=4$ so $|a|=2$. If it were to be $1$ it would mean that $a=1$ and it can't be $4$.
    Now $ab=c$ as $ab=1,a,b$ would all lead to trivial contradiction. Doing mere simple calculation we can find out the group table
    for this group which is,
        \[
    \begin{array}{c|cccc}
    \cdot & 1 & a & b & c \\ \hline
    1 & 1 & a & b & c \\
    a & a & 1 & c & b \\
    b & b & c & 1 & a \\
    c & c & b & a & 1
    \end{array}
\]
Since the table is symmetric its abelian.
\end{proof}
