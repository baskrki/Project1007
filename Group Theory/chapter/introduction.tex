\section{Introduction}

\begin{problem}
    Determine which of the following binary operations are associative:
    \begin{enumerate}
        \item the operation on $\ZZ$ defined by $a*b = a - b$
        \item the operation on $\RR$ defined by $a*b = a + b + ab$
        \item the operation on $\QQ$ defined by $a*b = \frac{a+b}{5}$
        \item the operation on $\ZZ \times \ZZ$ defined by $(a,b)*(c,d) = (ad+bc, bd)$
        \item the operation on $\QQ-\{0\}$ defined by $a*b = \frac{a}{b}$
    \end{enumerate}
\end{problem}

\begin{proof}
    This exercise is pretty trivial as we just have to compute $a*(b*c)=(a*b)*c$.
    For $1.$, $a*(b*c)=a*(b-c)=a-(b-c)=a-b+c$ but $(a*b)*c=(a-b)*c=(a-b)-c=a-b-c$.
    Rest of them are pretty similar.
\end{proof}

\begin{problem}
    Prove that addition of residue classes in $\ZZ/n\ZZ$ is associative.
\end{problem}

\begin{proof}
    By definition of residue addition,
    \[ [a]+[b]= [a+b] \]
    Therefore, we get the following.
    \begin{align*}
        [a]+([b]+[c])&=[a]+[b+c]\\
        &= [a+(b+c)] \\
        &= [(a+b)+c] \\
        &= [a+b]+[c] \\
        &= ([a]+[b])+[c]
    \end{align*}
\end{proof}

\begin{problem}
    Prove that multiplication of residue classes in $\ZZ/n\ZZ$ is associative.
\end{problem}

\begin{proof}
     By definition of residue multiplication,
    \[ [a]\cdot[b]= [a \cdot b] \]
    Therefore, we get the following.
    \begin{align*}
        [a]\cdot ([b] \cdot [c]) &= [a]\cdot [b \cdot c] \\
        &= [a \cdot (b \cdot c) ] \\
        &= [(a \cdot b) \cdot c] \\
        &= [a \cdot b] \cdot [c] \\
        &= ([a] \cdot [b]) \cdot [c]
    \end{align*}
\end{proof}

\begin{problem}
    Prove for all $n > 1$ that $\ZZ/n\ZZ$ is not a group under multiplication of residue classes.
\end{problem}

\begin{proof}
    There is no multiplicative inverse for $[a]$ with $\gcd(a,n)>1$. The proof for it pretty trivial,
    if there was a inverse, it would mean that
    \[ a \cdot b \equiv 1 \pmod{n} \implies 0 \equiv 1 \pmod{\gcd(a,n)} \]
    That's a contradiction. 
\end{proof}

\begin{problem}
    Determine which of the following sets are groups under addition:
    \begin{enumerate}
        \item the set of rational numbers (including $0 = 0/1$) in lowest terms whose denominators are odd
        \item the set of rational numbers (including $0 = 0/1$) in lowest terms whose denominators are even
        \item the set of rational numbers of absolute value $< 1$
        \item the set of rational numbers of absolute value $\ge 1$ together with 0
        \item the set of rational numbers with denominators equal to 1 or 2
        \item the set of rational numbers with denominators equal to 1, 2 or 3.
    \end{enumerate}
\end{problem}

\begin{proof}
    Below are the answers: 
    \begin{enumerate}
        \item Group.
        \item Not a group. Because it is not closed. (Eg: $5/2 + 7/2 = 6$)
        \item Not a group. Because again it is not closed. (Eg : $0.99+0.98=1.97$)
        \item Not a group. Because again it is not closed. (Eg : $4+(-3.1)=0.9$)
        \item Group.
        \item Not a group. Because its not closed. (Eg : $1/2+1/3=5/6$)
    \end{enumerate}
\end{proof}

\begin{problem}
    Let $G = \{x \in \mathbb{R} \mid 0 \le x < 1\}$ and for $x, y \in G$ let $x*y$ be the fractional part of $x+y$ 
    (i.e., $x*y = x+y - \floor{x+y}$ where $\floor{a}$ is the greatest integer less than or equal to $a$).
    Prove that $*$ is a well defined binary operation on $G$ and that $G$ is an abelian group under $*$ (called the real numbers mod 1).
\end{problem}

\begin{proof}
    It is well defined as,
    \[ x_1 = x_2, y_1 = y_2 \implies x_1*y_1 = x_1+y_1-\floor{x_1+y_1}\]
    \[ \implies x_1+y_1-\floor{x_1+y_1}=x_2+y_2 - \floor{x_2+y_2} = x_2 * y_2 \]
    Also, its closed as $x*y = \{x+y\}$ and $0 \le {x+y} < 1$. One can check that inverses and identity exists. For associativity,
    \begin{align*}
        x*(y*z) &= x*(y+z-\floor{y+z}) \\
        &= \{x+y+z - \floor{y+z}\} \\
        &= \{ x+y+z \} \\
        &= \{ x+y-\floor{x+y}+z \} \\
        &= (x*y)*z
    \end{align*}
\end{proof}
